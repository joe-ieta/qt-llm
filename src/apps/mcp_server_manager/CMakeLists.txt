find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core Gui Widgets Network)

file(GLOB_RECURSE MCP_SERVER_MANAGER_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp"
)
list(FILTER MCP_SERVER_MANAGER_SOURCES EXCLUDE REGEX "/(release|debug)/")
list(FILTER MCP_SERVER_MANAGER_SOURCES EXCLUDE REGEX "/moc_.*\\.cpp$")

file(GLOB_RECURSE MCP_SERVER_MANAGER_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/*.h"
)
list(FILTER MCP_SERVER_MANAGER_HEADERS EXCLUDE REGEX "/(release|debug)/")
list(FILTER MCP_SERVER_MANAGER_HEADERS EXCLUDE REGEX "/moc_.*\\.h$")

add_executable(mcp_server_manager
    ${MCP_SERVER_MANAGER_SOURCES}
    ${MCP_SERVER_MANAGER_HEADERS}
)

target_link_libraries(mcp_server_manager
    PRIVATE
    QtLlm::Conversation
    Qt${QT_VERSION_MAJOR}::Core
    Qt${QT_VERSION_MAJOR}::Gui
    Qt${QT_VERSION_MAJOR}::Widgets
    Qt${QT_VERSION_MAJOR}::Network
)
